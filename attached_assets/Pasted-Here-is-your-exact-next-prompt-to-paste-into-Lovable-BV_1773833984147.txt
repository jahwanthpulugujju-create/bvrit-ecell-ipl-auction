```typescript
async function confirmSale() {
  const pending = auctionState?.pending_sale;
  if (!pending) return;

  const team = teams.find(t => t.id === pending.team_id);
  if (!team || team.purse < pending.amount) {
    showToast('ERROR: Team cannot afford this player', 'error');
    return;
  }

  // Check RTM conditions
  const player = players.find(p => p.id === pending.player_id);
  const rtmConfig = await fetchConfig();
  
  if (
    player?.previous_team_id &&
    rtmConfig.rtm_enabled &&
    player.previous_team_id !== pending.team_id
  ) {
    const prevTeam = teams.find(t => t.id === player.previous_team_id);
    if (prevTeam && prevTeam.rtm_remaining > 0 && prevTeam.purse >= pending.amount) {
      // Trigger RTM window
      await supabase.from('rtm_state').update({
        active: true,
        player_id: pending.player_id,
        winning_team_id: pending.team_id,
        eligible_team_id: player.previous_team_id,
        matched_price: pending.amount,
        timer_expires_at: Date.now() + (rtmConfig.rtm_window_seconds * 1000),
        decision: null
      }).eq('id', (await supabase.from('rtm_state').select('id').limit(1).single()).data!.id);

      await supabase.from('auction_state').update({ status: 'rtm_window' })
        .eq('id', auctionState!.id);
      return;
    }
  }

  // No RTM — finalize directly
  await finalizeSale(pending.player_id, pending.team_id, pending.amount);
}
```

```typescript
async function finalizeSale(playerId: string, teamId: string, amount: number) {
  const team = teams.find(t => t.id === teamId);
  const player = players.find(p => p.id === playerId);
  if (!team || !player) return;

  const newPurse = team.purse - amount;
  const newSquad = [...(team.squad || []), playerId];
  const roleCounts = { ...team.role_counts };
  
  // Update role count
  const roleKey = player.role.replace('-', '_') as keyof typeof roleCounts;
  roleCounts[roleKey] = (roleCounts[roleKey] || 0) + 1;

  // Batch all writes
  await Promise.all([
    supabase.from('players').update({
      status: 'sold',
      sold_to_team_id: teamId,
      sold_price: amount
    }).eq('id', playerId),

    supabase.from('teams').update({
      purse: newPurse,
      squad: newSquad,
      role_counts: roleCounts
    }).eq('id', teamId),

    supabase.from('auction_state').update({
      status: 'live',
      pending_sale: null,
      current_player_id: null,
      current_bid_amount: 0,
      current_bid_team_id: null,
      leading_team_id: null,
      timer_running: false,
      last_sold: { player_id: playerId, team_id: teamId, amount, timestamp: Date.now() }
    }).eq('id', auctionState!.id),

    supabase.from('team_player_freezes').delete().eq('player_id', playerId),

    supabase.from('auction_log').insert({
      type: 'sold',
      player_id: playerId,
      team_id: teamId,
      amount,
      actor: 'admin',
      message: `${player.name} sold to ${team.name} for ₹${(amount/100).toFixed(1)} Cr`,
      timestamp: Date.now()
    })
  ]);

  playSoldFanfare();
}
```

## STEP 8: Admin — RTM Decision Listener

```typescript
useEffect(() => {
  if (!auctionState || auctionState.status !== 'rtm_window') return;

  const channel = supabase.channel('rtm-decision-watch')
    .on('postgres_changes', {
      event: 'UPDATE',
      schema: 'public',
      table: 'rtm_state'
    }, async (payload) => {
      const rtm = payload.new as RtmState;
      if (!rtm.decision) return;

      if (rtm.decision === 'accepted') {
        // RTM used — finalize sale with previous team
        await finalizeSale(rtm.player_id, rtm.eligible_team_id, rtm.matched_price);
        // Decrement RTM cards
        const eligibleTeam = teams.find(t => t.id === rtm.eligible_team_id);
        if (eligibleTeam) {
          await supabase.from('teams').update({
            rtm_remaining: Math.max(0, eligibleTeam.rtm_remaining - 1)
          }).eq('id', rtm.eligible_team_id);
        }
        // Log RTM usage
        await supabase.from('rtm_usage').insert({
          team_id: rtm.eligible_team_id,
          player_id: rtm.player_id,
          matched_price: rtm.matched_price,
          timestamp: Date.now()
        });
        showToast(`🟣 RTM used! ${eligibleTeam?.name} claimed player for ₹${(rtm.matched_price/100).toFixed(1)} Cr`, 'rtm');
      } else {
        // Declined or expired — finalize with original winner
        await finalizeSale(rtm.player_id, rtm.winning_team_id, rtm.matched_price);
        showToast('RTM declined — original winner confirmed', 'success');
      }

      // Clear RTM state
      await supabase.from('rtm_state').update({
        active: false,
        decision: null,
        player_id: null,
        winning_team_id: null,
        eligible_team_id: null
      }).eq('id', rtm.id);
    })
    .subscribe();

  return () => supabase.removeChannel(channel);
}, [auctionState?.status, teams]);
```
