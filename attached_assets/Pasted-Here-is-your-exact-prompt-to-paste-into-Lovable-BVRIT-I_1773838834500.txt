```typescript
// In the admin import UI:
const PREVIEW_ROWS = 20;
const BATCH_SIZE = 50; // Insert in batches of 50 to avoid Supabase payload limits

async function importPlayersFromCSV(parsedRows: ParsedPlayer[]) {
  const total = parsedRows.length;
  let imported = 0;
  let failed = 0;
  
  setImportProgress({ total, imported: 0, failed: 0, status: 'running' });

  // Process in batches
  for (let i = 0; i < parsedRows.length; i += BATCH_SIZE) {
    const batch = parsedRows.slice(i, i + BATCH_SIZE);
    
    const { error } = await supabase.from('players').upsert(
      batch.map(p => ({
        ...p,
        status: 'available',
        source: 'imported',
        import_batch_id: currentImportBatchId
      })),
      { onConflict: 'name,franchise', ignoreDuplicates: false }
    );

    if (error) {
      console.error('Batch import error:', error);
      failed += batch.length;
    } else {
      imported += batch.length;
    }

    setImportProgress({ total, imported, failed, status: 'running' });
    
    // Small delay between batches to avoid rate limiting
    await new Promise(resolve => setTimeout(resolve, 100));
  }

  setImportProgress({ total, imported, failed, status: 'complete' });
  showToast(`✓ Imported ${imported} players (${failed} skipped)`, imported > 0 ? 'success' : 'error');
}
```

```typescript
function useIntersectionObserver(ref: React.RefObject<Element>) {
  const [isVisible, setIsVisible] = useState(false);
  
  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => { if (entry.isIntersecting) setIsVisible(true); },
      { rootMargin: '200px' } // Start rendering 200px before entering viewport

}
```

```typescript
async function autoFillQueue() {
  const available = players.filter(p => p.status === 'available');
  const ordered = [
    ...available.filter(p => p.category === 'marquee').sort((a, b) => b.rating - a.rating),
    ...available.filter(p => p.category === 'premium').sort((a, b) => b.rating - a.rating),
    ...available.filter(p => p.category === 'mid-tier').sort((a, b) => b.rating - a.rating),
    ...available.filter(p => p.category === 'budget').sort((a, b) => b.rating - a.rating),
  ];

  await supabase.from('auction_state').update({
    auction_queue: ordered.map(p => p.id)
  }).eq('id', auctionState!.id);

  showToast(`✓ Queue filled with ${ordered.length} players`, 'success');
}

async function shuffleQueue() {
  const current = [...(auctionState?.auction_queue || [])];
  // Fisher-Yates shuffle
  for (let i = current.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [current[i], current[j]] = [current[j], current[i]];
  }
  await supabase.from('auction_state').update({ auction_queue: current })
    .eq('id', auctionState!.id);
  showToast('Queue shuffled', 'success');
}
```

```typescript
const ADMIN_TABLE_PAGE_SIZE = 50;
const [adminPage, setAdminPage] = useState(0);
const [adminTableSearch, setAdminTableSearch] = useState('');
const [adminTableTotal, setAdminTableTotal] = useState(0);

async function fetchAdminPlayers() {
  const from = adminPage * ADMIN_TABLE_PAGE_SIZE;
  const to = from + ADMIN_TABLE_PAGE_SIZE - 1;

  let query = supabase.from('players').select('*', { count: 'exact' });

  if (adminTableSearch) {
    query = query.ilike('name', `%${adminTableSearch}%`);
  }

  const { data, count, error } = await query
    .order('category')
    .order('name')
    .range(from, to);

  if (!error) {
    setAdminPlayers(data || []);
    setAdminTableTotal(count || 0);
  }
}

useEffect(() => { fetchAdminPlayers(); }, [adminPage, adminTableSearch]);
```
