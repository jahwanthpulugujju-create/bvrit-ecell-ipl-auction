**Freeze banner logic:**
```typescript
function handleFreezeChange(payload) {
  const freeze = payload.new;
  if (freeze.team_id !== myTeamId) return;
  if (freeze.player_id !== currentPlayerId) return;
  
  const remaining = Math.max(0, freeze.freeze_expires_at - Date.now());
  if (remaining > 0) {
    setFreezeRemaining(remaining);
    setShowFreezeBanner(true);
    // Start local countdown — no extra Supabase reads
    const interval = setInterval(() => {
      const rem = Math.max(0, freeze.freeze_expires_at - Date.now());
      setFreezeRemaining(rem);
      if (rem <= 0) {
        setShowFreezeBanner(false);
        clearInterval(interval);
      }
    }, 100);
  }
}
```

```typescript
function handleDisplayFreezeUpdate(payload) {
  const freeze = payload.new;
  const teamId = freeze.team_id;
  
  setTeamFreezes(prev => ({ ...prev, [teamId]: freeze }));
  
  // Local countdown — compute from freeze_expires_at every 1s
  // No additional Supabase reads
  const interval = setInterval(() => {
    const rem = Math.max(0, freeze.freeze_expires_at - Date.now());
    setTeamFreezeRemaining(prev => ({ ...prev, [teamId]: rem }));
    if (rem <= 0) {
      setTeamFreezes(prev => { const n = {...prev}; delete n[teamId]; return n; });
      clearInterval(interval);
    }
  }, 1000);
}
```
