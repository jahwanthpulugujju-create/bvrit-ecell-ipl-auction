```typescript
async function registerBid(teamId: string): Promise<BidResult> {
  const team = teams.find(t => t.id === teamId);
  const state = auctionState;
  if (!team || !state || !state.current_player_id) return { success: false, reason: 'NO_CURRENT_PLAYER' };

  const now = Date.now();

  // 1. Check per-player freeze
  const freeze = freezes.find(f =>
    f.team_id === teamId &&
    f.player_id === state.current_player_id &&
    f.freeze_expires_at > now
  );
  if (freeze) {
    const remaining = Math.ceil((freeze.freeze_expires_at - now) / 1000);
    return { success: false, reason: 'PLAYER_COOLDOWN', remainingSeconds: remaining };
  }

  // 2. Check global cooldown
  const global = globalCooldowns[teamId];
  if (global && global.global_expires_at > now) {
    const remaining = Math.ceil((global.global_expires_at - now) / 1000);
    return { success: false, reason: 'GLOBAL_COOLDOWN', remainingSeconds: remaining };
  }

  // 3. Compute new bid amount
  const increment = selectedIncrement; // from context state
  const newBid = state.current_bid_amount + increment;

  // 4. Validate purse
  if (team.purse < newBid) {
    return { success: false, reason: 'INSUFFICIENT_PURSE', purseRemaining: team.purse };
  }

  // 5. Compute freeze duration
  const freezeSecs = computeFreezeSecs(newBid);
  const freezeExpiresAt = now + (freezeSecs * 1000);
  const globalExpiresAt = now + 1000; // 1s global cooldown

  // 6. Atomic batch write to Supabase
  const bidId = crypto.randomUUID();
  const { error } = await supabase.rpc('register_bid', {
    p_bid_id: bidId,
    p_player_id: state.current_player_id,
    p_team_id: teamId,
    p_amount: newBid,
    p_timer_expires_at: now + (timerConfig.bid_reset_seconds * 1000),
    p_freeze_expires_at: freezeExpiresAt,
    p_freeze_seconds: freezeSecs,
    p_global_expires_at: globalExpiresAt
  });

  if (error) return { success: false, reason: 'SUPABASE_ERROR', error };

  // 7. Play sound
  playBidBeep();

  return { success: true, freezeSeconds: freezeSecs };
}
```

```sql
CREATE OR REPLACE FUNCTION register_bid(
  p_bid_id uuid,
  p_player_id uuid,
  p_team_id uuid,
  p_amount integer,
  p_timer_expires_at bigint,
  p_freeze_expires_at bigint,
  p_freeze_seconds integer,
  p_global_expires_at bigint
) RETURNS void AS $$
BEGIN
  -- Insert bid
  INSERT INTO bids (id, player_id, team_id, amount, timestamp)
  VALUES (p_bid_id, p_player_id, p_team_id, p_amount, extract(epoch from now()) * 1000);

  -- Update auction state
  UPDATE auction_state SET
    current_bid_amount = p_amount,
    current_bid_team_id = p_team_id,
    leading_team_id = p_team_id,
    timer_expires_at = p_timer_expires_at,
    timer_running = true
  WHERE id = (SELECT id FROM auction_state LIMIT 1);

  -- Upsert player freeze
  INSERT INTO team_player_freezes (team_id, player_id, freeze_expires_at, freeze_seconds, bid_amount, created_at)
  VALUES (p_team_id, p_player_id, p_freeze_expires_at, p_freeze_seconds, p_amount, extract(epoch from now()) * 1000)
  ON CONFLICT (team_id, player_id) DO UPDATE SET
    freeze_expires_at = EXCLUDED.freeze_expires_at,
    freeze_seconds = EXCLUDED.freeze_seconds,
    bid_amount = EXCLUDED.bid_amount;

  -- Upsert global cooldown
  INSERT INTO team_global_cooldowns (team_id, global_expires_at, last_bid_at)
  VALUES (p_team_id, p_global_expires_at, extract(epoch from now()) * 1000)
  ON CONFLICT (team_id) DO UPDATE SET
    global_expires_at = EXCLUDED.global_expires_at,
    last_bid_at = EXCLUDED.last_bid_at;

  -- Log the bid
  INSERT INTO auction_log (type, player_id, team_id, amount, actor, message, timestamp)
  VALUES ('bid', p_player_id, p_team_id, p_amount, 'admin',
    'Bid registered. Freeze: ' || p_freeze_seconds || 's', extract(epoch from now()) * 1000);
END;
$$ LANGUAGE plpgsql;
```

```typescript
function computeFreezeSecs(bidAmountLakhs: number): number {
  const config = freezeConfig; // from /config table
  const increments = Math.floor(bidAmountLakhs / config.increment_unit_lakhs);
  const raw = config.base_freeze_seconds + increments * config.increment_seconds;
  return Math.min(Math.max(config.min_freeze_seconds, raw), config.max_freeze_seconds);
}
```

```typescript
          ? prev.filter(t => t.id !== p.old.id)
          : prev.map(t => t.id === p.new.id ? p.new as Team : t).concat(
              prev.find(t => t.id === p.new.id) ? [] : [p.new as Team])))
      .subscribe(),

    supabase.channel('ctx-players')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'players' },
        p => setPlayers(prev => prev.map(pl => pl.id === p.new?.id ? p.new as Player : pl)))
      .subscribe(),

    supabase.channel('ctx-freezes')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'team_player_freezes' },
        p => {
          if (p.eventType === 'DELETE') {
            setFreezes(prev => prev.filter(f => f.id !== p.old.id));
          } else {
            setFreezes(prev => {
              const existing = prev.findIndex(f => f.id === p.new.id);
              return existing >= 0
                ? prev.map(f => f.id === p.new.id ? p.new as TeamPlayerFreeze : f)
                : [...prev, p.new as TeamPlayerFreeze];
            });
          }
        })
      .subscribe(),

    supabase.channel('ctx-globalcooldowns')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'team_global_cooldowns' },
        p => setGlobalCooldowns(prev => ({ ...prev, [p.new.team_id]: p.new })))
      .subscribe(),

    supabase.channel('ctx-rtm')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'rtm_state' },
        p => setRtmState(p.new as RtmState))
      .subscribe(),
  ];

  return () => channels.forEach(c => supabase.removeChannel(c));
```

```tsx
function TeamBidButton({ team }: { team: Team }) {

  const currentPlayerId = auctionState?.current_player_id;

  // Local countdown for freeze ring — reads from freezes array, no extra Supabase calls
  useEffect(() => {
    if (!currentPlayerId) return;
    const freeze = freezes.find(f =>
      f.team_id === team.id &&
      f.player_id === currentPlayerId &&
      f.freeze_expires_at > Date.now()
    );
    if (!freeze) { setLocalFreezeMs(0); return; }

    const tick = setInterval(() => {
      const rem = Math.max(0, freeze.freeze_expires_at - Date.now());
      setLocalFreezeMs(rem);
      if (rem <= 0) {
        clearInterval(tick);
        setJustExpired(true);
        setTimeout(() => setJustExpired(false), 200);
      }
    }, 100);
    return () => clearInterval(tick);
  }, [freezes, team.id, currentPlayerId]);

  // Local countdown for global cooldown
  useEffect(() => {
    const gc = globalCooldowns[team.id];
    if (!gc || gc.global_expires_at <= Date.now()) { setGlobalMs(0); return; }
    const tick = setInterval(() => {
      const rem = Math.max(0, gc.global_expires_at - Date.now());
      setGlobalMs(rem);
      if (rem <= 0) clearInterval(tick);
    }, 100);
    return () => clearInterval(tick);
  }, [globalCooldowns, team.id]);

  const isFrozen = localFreezeMs > 0;
  const isGlobal = globalMs > 0;
  const isDisabled = isFrozen || isGlobal;

  // Freeze ring: SVG circle, stroke-dashoffset tracks remaining %
  const freeze = freezes.find(f => f.team_id === team.id && f.player_id === currentPlayerId);

      showToast(`Bid accepted. ${team.name}'s freeze: ${result.freezeSeconds}s.`, 'success');
```

```tsx
function TeamAuthModal({ team, onAuthenticated, onClose, attempts, lockUntil }: AuthModalProps) {

  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState('');
  const [shaking, setShaking] = useState(false);

  const isLocked = lockRemaining > 0;

  function handleSubmit() {
    if (isLocked) return;
    const hash = btoa(`${team.slug}:${password}`);
    if (hash === team.password_hash) {
      sessionStorage.setItem(`team_auth_${team.slug}`, 'authenticated');
      onAuthenticated();

      setShaking(true);

    }

            </div>
          </>
        )}
      </div>
    </div>
  );
}
```

```tsx
// Zone 4 — Team Budgets Panel
function TeamBudgetsPanel() {
  const { teams, auctionState, freezes } = useAuction();
  const [teamFreezeRemaining, setTeamFreezeRemaining] = useState<Record<string, number>>({});

  // Local countdown for all team freezes — no extra Supabase reads
  useEffect(() => {
    const tick = setInterval(() => {
      const updated: Record<string, number> = {};
      if (!auctionState?.current_player_id) return;
      freezes.forEach(f => {
        if (f.player_id !== auctionState.current_player_id) return;
        const rem = Math.max(0, f.freeze_expires_at - Date.now());
        if (rem > 0) updated[f.team_id] = rem;
      });
      setTeamFreezeRemaining(updated);
    }, 1000);
    return () => clearInterval(tick);
  }, [freezes, auctionState?.current_player_id]);

  const sortedTeams = [...teams]
    .filter(t => t.is_active)
    .sort((a, b) => b.purse - a.purse);

  return (
    <div className="team-budgets-panel">
      <h3 className="panel-header">TEAM BUDGETS</h3>
      {sortedTeams.map(team => {
        const isLeading = team.id === auctionState?.leading_team_id;
        const freezeMs = teamFreezeRemaining[team.id];
        const pct = team.initial_purse > 0 ? (team.purse / team.initial_purse) * 100 : 0;
        const barColor = pct > 60 ? 'var(--accent-emerald)' : pct > 20 ? 'var(--accent-gold)' : 'var(--accent-crimson)';

        return (
          <div
            key={team.id}
            className={`budget-row ${isLeading ? 'leading' : ''}`}
            style={{ '--team-color': team.color } as React.CSSProperties}
          >
            <div className="budget-row-top">
              <span className="budget-team-name" style={{ color: isLeading ? 'var(--accent-primary)' : 'var(--text-primary)' }}>
                {team.name}
              </span>
              {freezeMs && freezeMs > 0 && (
                <span className="display-freeze-lock">
                  🔒 {Math.ceil(freezeMs / 1000)}s
                </span>
              )}
              <span className="budget-purse mono">₹{(team.purse/100).toFixed(0)} Cr</span>
            </div>
            <div className="budget-bar-track">
              <div
                className="budget-bar-fill"
                style={{ width: `${pct}%`, background: barColor }}
              />
            </div>
          </div>
        );
      })}
    </div>
  );
}
```

```typescript
// src/lib/sounds.ts
let ctx: AudioContext | null = null;

function getCtx(): AudioContext {
  if (!ctx) ctx = new AudioContext();
  return ctx;
}

function isMuted(): boolean {
  return localStorage.getItem('auction_muted') === 'true';
}

export function playBidBeep() {
  if (isMuted()) return;
  const c = getCtx();
  const osc = c.createOscillator();
  const gain = c.createGain();
  osc.connect(gain); gain.connect(c.destination);
  osc.frequency.value = 880;
  osc.type = 'sine';
  gain.gain.setValueAtTime(0.3, c.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, c.currentTime + 0.08);
  osc.start(); osc.stop(c.currentTime + 0.08);
}

export function playFreezeRejected() {
  if (isMuted()) return;
  const c = getCtx();
  const osc = c.createOscillator();
  const gain = c.createGain();
  osc.connect(gain); gain.connect(c.destination);
  osc.frequency.value = 440;
  osc.type = 'sawtooth';
  gain.gain.setValueAtTime(0.2, c.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, c.currentTime + 0.06);
  osc.start(); osc.stop(c.currentTime + 0.06);
}

export function playSoldFanfare() {
  if (isMuted()) return;
  const c = getCtx();
  const notes = [523.25, 659.25, 783.99, 1046.50]; // C5 E5 G5 C6
  notes.forEach((freq, i) => {
    const osc = c.createOscillator();
    const gain = c.createGain();
    osc.connect(gain); gain.connect(c.destination);
    osc.frequency.value = freq;
    osc.type = 'sine';
    const start = c.currentTime + i * 0.17;
    gain.gain.setValueAtTime(0.4, start);
```
